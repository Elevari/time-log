📋 Full setup guide
1
Open Google Sheets and create a new spreadsheet. Leave the sheet name as Sheet1.
2
Click Extensions → Apps Script. Delete all existing code and paste this:
const SHEET = 'Sheet1';

function doGet(e) {
  const action = e.parameter.action;
  if (action === 'getAll') return getAll();
  return res({error:'unknown action'});
}

function doPost(e) {
  const d = JSON.parse(e.postData.contents);
  if (d.action === 'add')    return addRow(d.entry);
  if (d.action === 'update') return updateRow(d.entry);
  if (d.action === 'delete') return deleteRow(d.id);
  return res({error:'unknown action'});
}

function getAll() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName(SHEET) || ss.insertSheet(SHEET);
  const rows = sh.getDataRange().getValues().slice(1);
  const entries = rows.map(r => ({
    id: r[0], className: r[1], date: r[2],
    clockIn: r[3], clockOut: r[4], createdAt: r[5]
  }));
  return res({entries});
}

function addRow(entry) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName(SHEET) || ss.insertSheet(SHEET);
  if (sh.getLastRow() === 0)
    sh.appendRow(['id','className','date',
                  'clockIn','clockOut','createdAt']);
  sh.appendRow([entry.id, entry.className, entry.date,
                entry.clockIn, entry.clockOut, entry.createdAt]);
  return res({ok:true});
}

function updateRow(entry) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName(SHEET);
  const data = sh.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === entry.id) {
      sh.getRange(i+1,1,1,6).setValues([[
        entry.id, entry.className, entry.date,
        entry.clockIn, entry.clockOut, entry.createdAt
      ]]);
      return res({ok:true});
    }
  }
  return res({error:'not found'});
}

function deleteRow(id) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName(SHEET);
  const data = sh.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === id) {
      sh.deleteRow(i+1); return res({ok:true});
    }
  }
  return res({error:'not found'});
}

function res(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

1
Go to github.com and create a free account if you don't have one.
2
Click New repository. Name it something like time-tracker. Set it to Public. Click Create repository.
3
You need three files in your repo. Download or create each one:
4
index.html — this app file (save this page as index.html).

manifest.json — create this file with the content below: 
{
  "name": "Time Tracker",
  "short_name": "Time.Log",
  "description": "Instructor time tracker",
  "start_url": "/",
  "display": "standalone",
  "background_color": "#0e0e11",
  "theme_color": "#0e0e11",
  "icons": [
    { "src": "icon-192.png", "sizes": "192x192", "type": "image/png" },
    { "src": "icon-512.png", "sizes": "512x512", "type": "image/png" }
  ]
}
sw.js — create this file (service worker for offline use): 
const CACHE = 'timelog-v1';
const ASSETS = ['/', '/index.html', '/manifest.json'];
self.addEventListener('install', e =>
  e.waitUntil(caches.open(CACHE).then(c => c.addAll(ASSETS)))
);
self.addEventListener('fetch', e =>
  e.respondWith(
    caches.match(e.request).then(r => r || fetch(e.request))
  )
);
5
Upload all three files to your GitHub repo using Add file → Upload files.
6
Go to repo Settings → Pages. Under Source, select main branch and click Save. Your app will be live at:
https://yourusername.github.io/time-tracker
