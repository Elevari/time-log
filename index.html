📖 Setup Instructions
Step 1 – Create Google Sheet
Open Google Sheets and create a new spreadsheet. Name it Instructor Time Tracker. Add these headers in row 1: Timestamp | Class Name | Date | Clock-In | Clock-Out | Total Hours | Notes Step 2 – Create Apps Script
In the sheet, go to Extensions → Apps Script. Replace all code with: 
const SHEET_NAME = 'Sheet1';

function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME);
  const data = JSON.parse(e.postData.contents);

  if (data.action === 'append') {
    sheet.appendRow([
      data.timestamp, data.className, data.date,
      data.clockIn, data.clockOut, data.totalHours, data.notes
    ]);
    return response({status:'ok'});
  }
  if (data.action === 'getAll') {
    const rows = sheet.getDataRange().getValues().slice(1);
    return response({status:'ok', data: rows});
  }
  return response({status:'error', msg:'Unknown action'});
}

function response(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  return response({status:'ok', msg:'Time Tracker API running'});
}
Step 3 – Deploy
Click Deploy → New Deployment. Select Web App. Set Execute as: Me and Who has access: Anyone. Copy the Web App URL and paste it above.

Step 4 – GitHub Pages
1. Create a new GitHub repository and push this index.html file.
2. Go to repo Settings → Pages, set source to main branch / root.
3. Your app will be live at https://yourusername.github.io/repo-name
